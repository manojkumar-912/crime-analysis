import React, { useState, useRef, useEffect } from 'react';
import axios from 'axios';
import { MessageSquare, X, Send, Bot, Terminal, ChevronRight } from 'lucide-react';
import { useData } from '../context/DataContext';

const ChatBot = () => {
    const [isOpen, setIsOpen] = useState(false);
    const [messages, setMessages] = useState([
        { role: 'assistant', content: 'Cyber Sentry AI active. State your analytical query.' }
    ]);
    const [input, setInput] = useState('');
    const [loading, setLoading] = useState(false);
    const { crimes } = useData();
    const chatEndRef = useRef(null);

    const scrollToBottom = () => {
        chatEndRef.current?.scrollIntoView({ behavior: 'smooth' });
    };

    useEffect(() => {
        scrollToBottom();
    }, [messages]);

    const handleSend = async (e) => {
        e.preventDefault();
        if (!input.trim()) return;

        const userMsg = { role: 'user', content: input };
        setMessages(prev => [...prev, userMsg]);
        setInput('');
        setLoading(true);

        try {
            const stats = { totalCrimes: crimes.length };
            const res = await axios.post('http://localhost:5000/api/chat', { 
                query: input,
                stats
            });
            setMessages(prev => [...prev, { role: 'assistant', content: res.data.response }]);
        } catch (err) {
            setMessages(prev => [...prev, { role: 'assistant', content: 'Connection to AI Engine lost. Check system logs.' }]);
        } finally {
            setLoading(false);
        }
    };

    const quickQueries = [
        "Risk Summary",
        "Top Hotspots",
        "System Help"
    ];

    return (
        <div className="fixed bottom-6 right-6 z-[10000]">
            {/* Chat Bubble */}
            {!isOpen && (
                <button 
                    onClick={() => setIsOpen(true)}
                    className="w-16 h-16 rounded-full bg-cyber-neon shadow-[0_0_20px_rgba(0,242,255,0.4)] flex items-center justify-center hover:scale-110 transition-all group"
                >
                    <div className="absolute -top-1 -right-1 w-4 h-4 bg-red-500 rounded-full border-2 border-cyber-dark animate-pulse" />
                    <MessageSquare size={28} className="text-black group-hover:rotate-12 transition-all" />
                </button>
            )}

            {/* Chat Window */}
            {isOpen && (
                <div className="glass-card w-[380px] h-[550px] flex flex-col !p-0 overflow-hidden border-cyber-neon/30 shadow-[0_0_50px_rgba(0,0,0,0.8)] animate-enter">
                    {/* Header */}
                    <div className="p-4 border-b border-white/10 bg-cyber-neon/10 flex justify-between items-center">
                        <div className="flex items-center gap-3">
                            <div className="p-2 bg-black rounded-lg border border-cyber-neon/30">
                                <Bot size={20} className="text-cyber-neon" />
                            </div>
                            <div>
                                <h3 className="text-xs font-black tracking-widest uppercase">CYBER_SENTRY_AI</h3>
                                <div className="flex items-center gap-1">
                                    <span className="w-1.5 h-1.5 rounded-full bg-green-500 animate-pulse" />
                                    <span className="text-[9px] text-cyber-neon font-bold">COGNITIVE_CORE_ACTIVE</span>
                                </div>
                            </div>
                        </div>
                        <button onClick={() => setIsOpen(false)} className="text-white/20 hover:text-white transition-colors">
                            <X size={20} />
                        </button>
                    </div>

                    {/* Messages */}
                    <div className="flex-1 overflow-y-auto p-4 space-y-4 custom-scrollbar bg-black/20">
                        {messages.map((m, idx) => (
                            <div key={idx} className={`flex ${m.role === 'user' ? 'justify-end' : 'justify-start'}`}>
                                <div className={`max-w-[80%] p-3 rounded-2xl text-xs leading-relaxed ${
                                    m.role === 'user' 
                                    ? 'bg-cyber-neon/20 border border-cyber-neon/30 text-white rounded-tr-none' 
                                    : 'bg-white/5 border border-white/10 text-white/80 rounded-tl-none'
                                }`}>
                                    {m.role === 'assistant' && <div className="text-[9px] font-black text-cyber-neon uppercase mb-1 flex items-center gap-1"><Terminal size={10}/> Response:</div>}
                                    {m.content}
                                </div>
                            </div>
                        ))}
                        {loading && (
                            <div className="flex justify-start">
                                <div className="bg-white/5 border border-white/10 p-3 rounded-2xl rounded-tl-none flex gap-1">
                                    <span className="w-1.5 h-1.5 bg-cyber-neon/40 rounded-full animate-bounce" />
                                    <span className="w-1.5 h-1.5 bg-cyber-neon/40 rounded-full animate-bounce [animation-delay:0.2s]" />
                                    <span className="w-1.5 h-1.5 bg-cyber-neon/40 rounded-full animate-bounce [animation-delay:0.4s]" />
                                </div>
                            </div>
                        )}
                        <div ref={chatEndRef} />
                    </div>

                    {/* Quick Queries */}
                    <div className="p-3 border-t border-white/10 flex gap-2 overflow-x-auto no-scrollbar">
                        {quickQueries.map(q => (
                            <button 
                                key={q}
                                onClick={() => { setInput(q); }}
                                className="whitespace-nowrap px-3 py-1.5 rounded-full border border-white/10 text-[9px] font-bold uppercase text-white/40 hover:text-cyber-neon hover:border-cyber-neon/40 transition-all"
                            >
                                {q}
                            </button>
                        ))}
                    </div>

                    {/* Input */}
                    <form onSubmit={handleSend} className="p-4 border-t border-white/10 bg-black/40 flex gap-2">
                        <input 
                            type="text" 
                            placeholder="Type analytical query..."
                            value={input}
                            onChange={(e) => setInput(e.target.value)}
                            className="flex-1 bg-white/5 border border-white/10 rounded-xl px-4 py-2 text-xs focus:outline-none focus:border-cyber-neon/40 transition-all"
                        />
                        <button type="submit" className="p-2 bg-cyber-neon rounded-xl text-black hover:scale-105 active:scale-95 transition-all">
                            <Send size={18} />
                        </button>
                    </form>
                </div>
            )}
        </div>
    );
};

export default ChatBot;

import React, { createContext, useState, useEffect, useContext } from 'react';
import axios from 'axios';

const DataContext = createContext();

export const DataProvider = ({ children }) => {
    const [crimes, setCrimes] = useState([]);
    const [hotspots, setHotspots] = useState([]);
    const [temporalData, setTemporalData] = useState([]);
    const [trends, setTrends] = useState([]);
    const [predictions, setPredictions] = useState([]);
    const [analyticsData, setAnalyticsData] = useState(null);
    const [timeDepthData, setTimeDepthData] = useState(null);
    const [loading, setLoading] = useState(true);

    const fetchData = async () => {
        setLoading(true);
        try {
            const [crimeRes, hotspotRes, temporalRes, trendRes, predRes, analyticsRes, timeDepthRes] = await Promise.all([
                axios.get('http://localhost:5000/api/crimes'),
                axios.get('http://localhost:5000/api/hotspots'),
                axios.get('http://localhost:5000/api/analysis/temporal'),
                axios.get('http://localhost:5000/api/analysis/hotspot-trends'),
                axios.get('http://localhost:5000/api/analysis/predictions'),
                axios.get('http://localhost:5000/api/analysis/advanced-stats'),
                axios.get('http://localhost:5000/api/analysis/time-depth')
            ]);
            setCrimes(crimeRes.data);
            setHotspots(hotspotRes.data);
            setTemporalData(temporalRes.data);
            setTrends(trendRes.data);
            setPredictions(predRes.data);
            setAnalyticsData(analyticsRes.data);
            setTimeDepthData(timeDepthRes.data);
        } catch (err) {
            console.error('Error fetching crime data:', err);
        } finally {
            setLoading(false);
        }
    };

    useEffect(() => {
        fetchData();
    }, []);

    return (
        <DataContext.Provider value={{ 
            crimes, hotspots, temporalData, trends, predictions, 
            analyticsData, timeDepthData, loading, fetchData 
        }}>
            {children}
        </DataContext.Provider>
    );
};

import joblib
import pandas as pd
import numpy as np
import sys
import json
from datetime import datetime

def predict():
    try:
        # Load model and encoders
        model = joblib.load('models/crime_model.joblib')
        le_type = joblib.load('models/le_type.joblib')
        le_neighbor = joblib.load('models/le_neighbor.joblib')
        loc_mapping = joblib.load('models/loc_mapping.joblib')
        
        # Get current time features
        now = datetime.now()
        hour = now.hour
        day_of_week = now.weekday()
        month = now.month
        
        # We predict for all crime types and find the most likely hotspots
        all_types = le_type.classes_
        results = []
        
        for crime_type in all_types:
            type_encoded = le_type.transform([crime_type])[0]
            
            # Predict probabilities for each neighborhood
            X = pd.DataFrame([[hour, day_of_week, month, type_encoded]], 
                             columns=['hour', 'day_of_week', 'month', 'type_encoded'])
            
            probs = model.predict_proba(X)[0]
            
            # Get top neighborhoods
            top_indices = np.argsort(probs)[-3:][::-1]
            for idx in top_indices:
                if probs[idx] > 0.1: # Threshold
                    neighbor = le_neighbor.inverse_transform([idx])[0]
                    loc = loc_mapping.get(neighbor, {'lat': 0, 'lng': 0})
                    results.append({
                        'neighborhood': neighbor,
                        'crime_type': crime_type,
                        'probability': round(float(probs[idx]) * 100, 2),
                        'lat': loc['lat'],
                        'lng': loc['lng']
                    })
        
        # Sort by probability and return top 5
        results = sorted(results, key=lambda x: x['probability'], reverse=True)[:5]
        print(json.dumps(results))
        
    except Exception as e:
        print(json.dumps({"error": str(e)}))

if __name__ == "__main__":
    predict()

# Advanced Risk Lexicon
RISK_KEYWORDS = {
    'high': ['gun', 'shot', 'shooting', 'weapon', 'armed', 'robbery', 'riot', 'explosive', 'attack', 'kill'],
    'medium': ['fight', 'theft', 'stolen', 'suspicious', 'danger', 'alert', 'police', 'chase', 'emergency'],
    'low': ['noisy', 'loitering', 'argument', 'unpleasant', 'crowded', 'traffic']
}

def analyze_sentiment(text):
    text = text.lower()
    risk_score = 0
    
    # Calculate Risk Score
    for level, words in RISK_KEYWORDS.items():
        for word in words:
            if word in text:
                if level == 'high': risk_score += 40
                elif level == 'medium': risk_score += 20
                elif level == 'low': risk_score += 10
    
    risk_score = min(risk_score, 100)
    
    # Simple Sentiment (Mock)
    # Higher risk usually means negative sentiment
    sentiment = "Negative" if risk_score > 30 else "Neutral"
    if risk_score > 70: sentiment = "Critical"
    
    return {
        "text": text,
        "risk_score": risk_score,
        "sentiment": sentiment,
        "is_threat": risk_score >= 80
    }

import pandas as pd
import numpy as np
from pymongo import MongoClient
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import LabelEncoder
import joblib
import os
from datetime import datetime

# MongoDB Connection
MONGO_URI = "mongodb://localhost:27017/"
DB_NAME = "crime-analysis"

def train():
    print("Fetching data from MongoDB...")
    client = MongoClient(MONGO_URI)
    db = client[DB_NAME]
    collection = db["crimes"]
    
    data = list(collection.find())
    if not data:
        print("No data found to train model.")
        return

    df = pd.DataFrame(data)
    
    # Feature Engineering
    print("Preprocessing data...")
    df['timestamp'] = pd.to_datetime(df['timestamp'])
    df['hour'] = df['timestamp'].dt.hour
    df['day_of_week'] = df['timestamp'].dt.dayofweek
    df['month'] = df['timestamp'].dt.month
    
    # Encode Categorical Data
    le_type = LabelEncoder()
    le_neighbor = LabelEncoder()
    
    df['type_encoded'] = le_type.fit_transform(df['type'])
    df['neighbor_encoded'] = le_neighbor.fit_transform(df['neighborhood'])
    
    # Features and Target
    # Goal: Predict the neighborhood based on time and type
    X = df[['hour', 'day_of_week', 'month', 'type_encoded']]
    y = df['neighbor_encoded']
    
    print(f"Training Random Forest on {len(df)} records...")
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X, y)
    
    # Save Model and Encoders
    print("Saving model and metadata...")
    os.makedirs('models', exist_ok=True)
    joblib.dump(model, 'models/crime_model.joblib')
    joblib.dump(le_type, 'models/le_type.joblib')
    joblib.dump(le_neighbor, 'models/le_neighbor.joblib')
    
    # Save location mapping for neighborhoods
    df['lat'] = df['location'].apply(lambda x: x.get('lat', 0))
    df['lng'] = df['location'].apply(lambda x: x.get('lng', 0))
    
    loc_mapping = df.groupby('neighborhood').agg({
        'lat': 'first',
        'lng': 'first'
    }).to_dict('index')
    joblib.dump(loc_mapping, 'models/loc_mapping.joblib')
    
    print("Training complete. Model saved in server/ml/models/")

if __name__ == "__main__":
    train()

        const crimes = await Crime.find();
        const now = new Date();
        const currentHour = now.getHours();

        const neighborhoodData = crimes.reduce((acc, crime) => {
            const nh = crime.neighborhood;
            if (!acc[nh]) acc[nh] = { count: 0, timeMatches: 0, highSeverity: 0, lat: crime.location.lat, lng: crime.location.lng };
            acc[nh].count++;
            
            const crimeHour = new Date(crime.timestamp).getHours();
            if (Math.abs(crimeHour - currentHour) <= 2) acc[nh].timeMatches++;
            if (crime.severity === 'High') acc[nh].highSeverity++;
            
            return acc;
        }, {});

        const predictions = Object.keys(neighborhoodData).map(nh => {
            const data = neighborhoodData[nh];
            const freqScore = Math.min(data.count / 5, 1) * 40;
            const timeScore = Math.min(data.timeMatches / 3, 1) * 40;
            const sevScore = Math.min(data.highSeverity / 2, 1) * 20;

                lng: data.lng,

        const hourly = await Crime.aggregate([
            { $group: { _id: { $hour: "$timestamp" }, count: { $sum: 1 } } },
            { $sort: { _id: 1 } }
        ]);

        const days = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"];
        const daily = await Crime.aggregate([
            { $group: { _id: { $dayOfWeek: "$timestamp" }, count: { $sum: 1 } } },
            { $sort: { _id: 1 } }
        ]);

        const seasonal = await Crime.aggregate([
            { $group: { _id: { $month: "$timestamp" }, count: { $sum: 1 } } },
            { $sort: { _id: 1 } }
        ]);

        res.json({
            hourly: Array.from({ length: 24 }, (_, i) => ({
                hour: `${i}:00`,
                count: hourly.find(h => h._id === i)?.count || 0
            })),
            daily: daily.map(d => ({
                day: days[d._id - 1],
                count: d.count
            })),
            seasonal: seasonal.map(s => ({
                month: new Date(2000, s._id - 1).toLocaleString('default', { month: 'short' }),
                count: s.count
            }))
        });

        res.status(500).json({ message: err.message });

router.get('/social/stream', async (req, res) => {
    const neighborhoods = ['Midtown', 'Lower Manhattan', 'Harlem', 'Brooklyn Heights'];
    const users = ['SafetyWatcher', 'NY_Local', 'Citizen_X', 'UrbanSentinel', 'DailyNews_Bot'];
    
    const posts = Array.from({ length: 15 }, () => {
        const text = MOCK_POSTS[Math.floor(Math.random() * MOCK_POSTS.length)];
        return {
            id: Math.random().toString(36).substr(2, 9),
            user: users[Math.floor(Math.random() * users.length)],
            text: text,
            neighborhood: neighborhoods[Math.floor(Math.random() * neighborhoods.length)],
            timestamp: new Date(Date.now() - Math.random() * 3600000)
        };
    });
    
    // Simple internal sentiment analysis for the feed
    const analyzedPosts = await Promise.all(posts.map(async post => {
        return new Promise((resolve) => {
            const python = spawn('python.exe', [path.join(__dirname, '../ml/sentiment_service.py')]);
            let result = '';
            
            python.stdin.write(JSON.stringify({ text: post.text }));
            python.stdin.end();
            
            python.stdout.on('data', (data) => {
                result += data.toString();
            });
            
            python.on('close', () => {
                try {
                    const analysis = JSON.parse(result);
                    resolve({ ...post, ...analysis });
                } catch (e) {
                    resolve({ ...post, risk_score: 0, sentiment: 'Neutral' });
                }
            });
        });
    }));
    
    res.json(analyzedPosts.sort((a, b) => b.timestamp - a.timestamp));
});
